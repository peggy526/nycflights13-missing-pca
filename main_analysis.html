compare_vars = c(
  "dep_delay", "arr_delay", "air_time",
  "distance", "temp", "dewp", "humid",
  "wind_speed", "precip", "visib"
)

compare_one = function(v) {
  x = flight_analysis[[v]][pressure_group == "Missing"]
  y = flight_analysis[[v]][pressure_group == "Observed"]
  x = x[!is.na(x)]; y = y[!is.na(y)]
  
  sd_pool = sqrt(((length(x) - 1) * var(x) + (length(y) - 1) * var(y)) /
                   (length(x) + length(y) - 2))
  
  data.frame(
    variable   = v,
    n_missing  = length(x),
    n_observed = length(y),
    mean_miss  = round(mean(x), 2),
    mean_obs   = round(mean(y), 2),
    med_miss   = round(median(x), 2),
    med_obs    = round(median(y), 2),
    SMD        = round((mean(x) - mean(y)) / sd_pool, 3),
    p_wilcox   = signif(wilcox.test(x, y)$p.value, 3)
  )
}


compare_table = do.call(rbind, lapply(compare_vars, compare_one))

interp_one = function(t, p) {
  ok = !is.na(p)
  if (sum(ok) < 2) return(rep(NA_real_, length(p)))
  approx(as.numeric(t)[ok], p[ok], xout = as.numeric(t), rule = 1)$y
}

impute_pressure = function(g) {
  site_off = tapply(g$pressure, g$origin, mean, na.rm = TRUE) - mean(g$pressure, na.rm = TRUE)
  g %>%
    group_by(origin) %>%
    mutate(p_int = interp_one(time_hour, pressure)) %>%
    group_by(time_hour) %>%
    mutate(p_other = mean(pressure, na.rm = TRUE)) %>%
    ungroup() %>%
    mutate(p_other = ifelse(is.nan(p_other), NA_real_, p_other),
           p_int   = ifelse(is.na(p_int), p_other + site_off[origin], p_int)) %>%
    group_by(origin) %>%
    tidyr::fill(p_int, .direction = "downup") %>%
    ungroup() %>%
    pull(p_int)
}
